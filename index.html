const PAGES = ["home","movies","figures","merch","community"];

const rendered = { movies:false, figures:false, merch:false };


function showPage(page){

  if (!PAGES.includes(page)) page = "home";

  document.querySelectorAll("main > .page").forEach(el => { el.hidden = (el.dataset.page !== page); });

  document.querySelectorAll(".nav-links a").forEach(a => { a.classList.toggle("active", a.dataset.nav === page); });

  if (page === "movies" && !rendered.movies){ renderMovies(); rendered.movies = true; }

  if (page === "figures" && !rendered.figures){ renderFigures("all"); rendered.figures = true; }

  if (page === "merch" && !rendered.merch){ renderMerch(); rendered.merch = true; }

  window.scrollTo(0, 0);

}


document.querySelectorAll("[data-nav]").forEach(a => {

  a.addEventListener("click", e => {

    e.preventDefault();

    const page = a.dataset.nav;

    closeMenu();

    if (location.hash.slice(1) === page) showPage(page);

    else location.hash = page;

  });

});

window.addEventListener("hashchange", () => showPage(location.hash.slice(1) || "home"));

showPage(location.hash.slice(1) || "home");

function renderMovies(){

  const grid = document.getElementById("moviesGrid");

  grid.innerHTML = MOVIES.map(m => {

    return '<article class="movie-card anim">'

      + '<div class="movie-poster">'

      + '<img src="' + m.poster + '" alt="Постер фильма ' + m.title + '" loading="lazy" onerror="this.style.display=\'none\'">'

      + '<span class="poster-year">' + m.year + '</span>'

      + '</div>'

      + '<div class="movie-body">'

      + '<span class="movie-type">' + m.type + '</span>'

      + '<h3>' + m.title + '</h3>'

      + '<p class="movie-desc">' + m.desc + '</p>'

      + '</div>'

      + '</article>';

  }).join("");

}


/* ===== ФИГУРКИ ===== */

function brandName(key){

  return { hasbro:"Hasbro", takara:"Takara Tomy", threezero:"threezero", yolopark:"YOLOPARK" }[key] || key;

}

function renderFigures(filter){

  const grid = document.getElementById("figuresGrid");

  const list = filter === "all" ? FIGURES : FIGURES.filter(f => f.brand === filter);

  grid.innerHTML = list.map(f => {

    const idx = FIGURES.indexOf(f);

    return '<article class="figure-card anim">'

      + '<div class="figure-image">' + figureArt(idx, f.accent) + '</div>'

      + '<span class="figure-brand">' + brandName(f.brand) + '</span>'

      + '<h3>' + f.name + '</h3>'

      + '<div class="figure-meta">' + f.line + ' · ' + f.note + '</div>'

      + '<div class="figure-footer">'

      + '<span class="figure-price">' + f.price + '</span>'

      + '<span class="badge-status ' + f.status + '">' + (f.status === "pre" ? "Предзаказ" : "В наличии") + '</span>'

      + '</div></article>';

  }).join("");

}

document.getElementById("figureFilters").addEventListener("click", e => {

  const btn = e.target.closest(".filter");

  if (!btn) return;

  document.querySelectorAll("#figureFilters .filter").forEach(b => b.classList.remove("active"));

  btn.classList.add("active");

  renderFigures(btn.dataset.filter);

});
